cmake_minimum_required(VERSION 3.25)
project(canview_boot_trust_test LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(Python3 3.12 REQUIRED COMPONENTS Interpreter)
get_filename_component(canview_boot_repo "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${canview_boot_repo}/cmake/CanviewWarnings.cmake")
set(canview_bootutil_source "${CANVIEW_MCUBOOT_ROOT}/boot/bootutil")
add_library(canview_boot_config INTERFACE)
target_include_directories(canview_boot_config INTERFACE
    "${canview_boot_repo}/firmware/communicator/stm32/bootloader/include"
    "${canview_boot_repo}/shared/ota/src" "${canview_boot_repo}/shared/interface")
include("${canview_boot_repo}/firmware/communicator/stm32/bootloader/trust.cmake")
if(TARGET canview_boot_identity)
    add_executable(boot-trust-probe ../test_stm32_boot_identity.c)
    target_include_directories(boot-trust-probe PRIVATE "${canview_bootutil_source}/include")
    target_link_libraries(boot-trust-probe PRIVATE canview_boot_identity canview_boot_config canview_warnings)
endif()
