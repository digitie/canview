cmake_minimum_required(VERSION 3.20)
project(canview_lvgl_regression LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(LVGL_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../.tools/lvgl-8.4.0" CACHE PATH "공식 LVGL 8.4.0 checkout")
if(NOT EXISTS "${LVGL_SOURCE_DIR}/lvgl.h")
  message(FATAL_ERROR "공식 LVGL 8.4.0 소스가 필요합니다: ${LVGL_SOURCE_DIR}")
endif()
file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS "${LVGL_SOURCE_DIR}/src/*.c")
add_library(lvgl_real STATIC ${LVGL_SOURCES})
target_include_directories(lvgl_real SYSTEM PUBLIC "${LVGL_SOURCE_DIR}")
target_include_directories(lvgl_real PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
target_compile_definitions(lvgl_real PUBLIC LV_CONF_INCLUDE_SIMPLE)
add_executable(canview_lvgl_regression test_lvgl.c)
target_link_libraries(canview_lvgl_regression PRIVATE lvgl_real)
if(MSVC)
  target_compile_options(lvgl_real PRIVATE /utf-8 /wd4819)
  target_compile_options(canview_lvgl_regression PRIVATE /utf-8 /W4 /WX /UNDEBUG)
  target_compile_definitions(canview_lvgl_regression PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(canview_lvgl_regression PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  target_link_libraries(canview_lvgl_regression PRIVATE m)
endif()
enable_testing()
add_test(NAME canview_lvgl_regression COMMAND canview_lvgl_regression)
set_tests_properties(canview_lvgl_regression PROPERTIES TIMEOUT 60)
